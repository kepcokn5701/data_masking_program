분류 기준 (Threshold)

# 최고 점수가 아래 조건을 충족해야 함
if 최고점수 >= max(2, 샘플수 × 0.3):
    return 해당 유형
else:
    return "없음"

최소 2개 이상 매칭되어야 하며,

💻 핵심 코드 분석

자동 감지 함수 detect_column_type() 의 실제 동작

def detect_column_type(series):
    # 1단계: 컬럼에서 최대 20개 비어있지 않은 값 샘플링
    sample = series.dropna().astype(str).head(20)

    # 2단계: 5가지 정규표현식 패턴 정의
    phone_pat = re.compile(r"0\d{1,2}[-\s]?\d{3,4}[-\s]?\d{4}")
    email_pat = re.compile(r"[^@\s]+@[^@\s]+\.[^@\s]+")
    rrn_pat   = re.compile(r"\d{6}[-\s]?\d{7}")
    addr_pat  = re.compile(r"(?:서울|부산|대구|...17개 시도)")
    name_pat  = re.compile(r"^[가-힣]{2,4}$")

    # 3단계: 각 샘플에 대해 5가지 패턴 매칭 → 점수 누적
    scores = {"이름": 0, "전화번호": 0, "이메일": 0, ...}
    for v in sample:
        if phone_pat.search(v):  scores["전화번호"] += 1
        if email_pat.search(v):  scores["이메일"]  += 1
        ... # 나머지 패턴도 동일하게 검사

    # 4단계: 최고 점수 유형 선택 (임계값 이상일 때만)
    best = max(scores, key=scores.get)
    return best if scores[best] >= max(2, len(sample) * 0.3) else "없음"
